cmake_minimum_required(VERSION 3.21)
project(comic-chat-xr VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(nlohmann_json REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
# Use the same Pango bitmap integration as the selected FLTK toolkit.
get_filename_component(_comic_toolkit "${FLTK_DIR}/../.." ABSOLUTE)
set(_comic_pc_path "$ENV{PKG_CONFIG_PATH}")
if(EXISTS "${_comic_toolkit}/lib/pkgconfig/pangocairo.pc")
 set(ENV{PKG_CONFIG_PATH} "${_comic_toolkit}/lib/pkgconfig:${_comic_pc_path}")
endif()
pkg_check_modules(PANGO REQUIRED IMPORTED_TARGET pangocairo)
set(ENV{PKG_CONFIG_PATH} "${_comic_pc_path}")
pkg_check_modules(MODEL_EGL REQUIRED IMPORTED_TARGET egl)
option(COMIC_CHAT_XR "Build native OpenXR scene presentation" ON)
if(COMIC_CHAT_XR)
 pkg_check_modules(MODEL_XR REQUIRED IMPORTED_TARGET openxr)
endif()
include(GNUInstallDirs)
include(CTest)
set(art "${CMAKE_CURRENT_BINARY_DIR}/art")
file(GLOB_RECURSE original_art CONFIGURE_DEPENDS "v2.5-beta-1-modern/comicart/*.avb" "v2.5-beta-1-modern/comicart/*.bgb" "v2.5-beta-1-modern/artpack1/*.avb")
add_custom_command(OUTPUT "${art}/catalog.json"
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/tools/import_art.py" "${CMAKE_CURRENT_SOURCE_DIR}/v2.5-beta-1-modern" "${art}"
 DEPENDS native/tools/import_art.py native/tools/parallax.py native/backdrop-layers.json ${original_art} VERBATIM)
add_custom_target(comic-art ALL DEPENDS "${art}/catalog.json")
add_library(comic-session STATIC native/src/domed.cpp)
target_include_directories(comic-session PUBLIC native/src)
target_link_libraries(comic-session PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
add_library(comic-core STATIC native/src/art.cpp native/src/document.cpp native/src/render.cpp native/src/scene.cpp native/src/irc.cpp)
target_include_directories(comic-core PUBLIC native/src)
target_link_libraries(comic-core PUBLIC comic-session PNG::PNG PkgConfig::PANGO PkgConfig::GIO nlohmann_json::nlohmann_json Threads::Threads)
add_library(comic-graphics STATIC native/graphics/gltf-model.cpp native/graphics/gltf-render.cpp native/graphics/model-egl.cpp native/graphics/model-xr.cpp native/graphics/model-loader.cpp native/graphics/model-controller-profiles.cpp native/graphics/object-interaction.cpp)
target_include_directories(comic-graphics PUBLIC native/graphics)
target_compile_definitions(comic-graphics PRIVATE FSVR_MODEL_XR=$<BOOL:${COMIC_CHAT_XR}>)
target_link_libraries(comic-graphics PUBLIC comic-session OpenGL::GL PkgConfig::MODEL_EGL Threads::Threads nlohmann_json::nlohmann_json)
if(COMIC_CHAT_XR)
 target_link_libraries(comic-graphics PUBLIC PkgConfig::MODEL_XR)
endif()
add_executable(comic-chat-xr native/src/main.cpp)
add_dependencies(comic-chat-xr comic-art)
target_compile_options(comic-chat-xr PRIVATE -Wall -Wextra)
target_include_directories(comic-chat-xr PRIVATE native/suite)
target_compile_definitions(comic-chat-xr PRIVATE COMIC_ART_BUILD="${art}" COMIC_ART_INSTALL="${CMAKE_INSTALL_FULL_DATADIR}/comic-chat-xr/art")
target_link_libraries(comic-chat-xr PRIVATE comic-core comic-graphics fltk::gl-shared fltk::fltk-shared fltk::images-shared PkgConfig::FONTCONFIG)
include(native/suite/desktop-menu.cmake)
de_xr_desktop_menu(comic-chat-xr)
install(TARGETS comic-chat-xr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY "${art}/" DESTINATION ${CMAKE_INSTALL_DATADIR}/comic-chat-xr/art)
install(FILES LICENSE native/third_party/LICENSE.native-graphics native/suite/NOTICE native/upstream.json DESTINATION ${CMAKE_INSTALL_DATADIR}/comic-chat-xr/licenses)
install(FILES native/comic-chat-xr.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications RENAME org.de_xr.ComicChat.desktop)
install(FILES native/comic-chat-xr.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
 add_executable(comic-tests native/tests/core.cpp)
 target_compile_options(comic-tests PRIVATE -UNDEBUG)
 target_link_libraries(comic-tests PRIVATE comic-core comic-graphics)
 add_dependencies(comic-tests comic-art)
 add_test(NAME comic-art-layout-protocol COMMAND comic-tests "${art}" "${CMAKE_CURRENT_BINARY_DIR}/qa")
 add_test(NAME comic-irc-transport COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/irc.py" $<TARGET_FILE:comic-chat-xr>)
 set_tests_properties(comic-irc-transport PROPERTIES TIMEOUT 25)
endif()
set_property(TARGET comic-chat-xr PROPERTY INSTALL_RPATH_USE_LINK_PATH TRUE)
if(BUILD_TESTING AND COMIC_CHAT_XR)
 add_library(comic-mock-openxr SHARED native/tests/mock-openxr.cpp)
 target_link_libraries(comic-mock-openxr PRIVATE OpenGL::GL PkgConfig::MODEL_XR)
 add_test(NAME comic-openxr-stereo COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/xr.py" $<TARGET_FILE:comic-chat-xr> $<TARGET_FILE:comic-mock-openxr>)
 set_tests_properties(comic-openxr-stereo PROPERTIES TIMEOUT 20)
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/gltf-assets/manifest.json"
 COMMAND $<TARGET_FILE:comic-chat-xr> --export-assets "${CMAKE_CURRENT_BINARY_DIR}/gltf-assets"
 DEPENDS comic-chat-xr "${art}/catalog.json" VERBATIM)
add_custom_target(comic-gltf-assets ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/gltf-assets/manifest.json")
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/gltf-assets/" DESTINATION ${CMAKE_INSTALL_DATADIR}/comic-chat-xr/gltf)
install(FILES native/third_party/README.md native/third_party/gltf-dependencies.json DESTINATION ${CMAKE_INSTALL_DATADIR}/comic-chat-xr/licenses)
install(FILES native/third_party/cgltf/cgltf.h DESTINATION ${CMAKE_INSTALL_DATADIR}/comic-chat-xr/licenses/cgltf)
install(FILES native/third_party/stb/stb_image.h DESTINATION ${CMAKE_INSTALL_DATADIR}/comic-chat-xr/licenses/stb)
if(BUILD_TESTING)
 add_test(NAME comic-original-art-format COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/art.py")
 find_program(COMIC_DOME_COMPOSITOR dome-compositor HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../DomeD/target/release")
 find_program(COMIC_XVFB Xvfb)
 find_program(COMIC_XDOTOOL xdotool)
 if(COMIC_DOME_COMPOSITOR AND COMIC_XVFB)
  add_test(NAME comic-wayland-menu COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/wayland.py" $<TARGET_FILE:comic-chat-xr> "${COMIC_DOME_COMPOSITOR}")
  set_tests_properties(comic-wayland-menu PROPERTIES TIMEOUT 50)
 endif()
 if(COMIC_XVFB AND COMIC_XDOTOOL)
  add_test(NAME comic-native-chat-gestures COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/ui.py" $<TARGET_FILE:comic-chat-xr>)
  set_tests_properties(comic-native-chat-gestures PROPERTIES TIMEOUT 50)
 endif()
endif()
if(BUILD_TESTING AND COMIC_CHAT_XR AND COMIC_DOME_COMPOSITOR AND COMIC_XVFB)
 add_test(NAME comic-dome-modes COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/native/tests/mode.py" $<TARGET_FILE:comic-chat-xr> "${COMIC_DOME_COMPOSITOR}" $<TARGET_FILE:comic-mock-openxr>)
 set_tests_properties(comic-dome-modes PROPERTIES TIMEOUT 90)
endif()
